Para calcular as reações de apoio, é necessário aplicar as equações de equilíbrio estático:

\begin{flalign*}
   & \sum F_y = 0                                           &  & \\
   & R_N + R_M - \num{53155.6} - \num{1580} - \num{540} = 0 &  &
\end{flalign*}

\begin{equation}
  R_N + R_M - \num{55275.6} = 0
  \label{eq:sum_fy_vl2}
\end{equation}

\vspace{1em}

Os momentos atuantes no ponto N são:

\begin{equation*}
  \num{2.437} \cdot R_M
\end{equation*}

\begin{equation*}
  -\num{1.2185} \cdot \num{53155.6} = \SI{-64770.1}{\newton\meter}
\end{equation*}

\begin{equation*}
  -\num{1.747} \cdot \num{1580} = \SI{-2760.26}{\newton\meter}
\end{equation*}

\begin{equation*}
  -\num{2.437} \cdot \num{540} = \SI{-1315.98}{\newton\meter}
\end{equation*}

\begin{flalign*}
   & \sum M_N = 0                                                                    &  & \\
   & M_N - M_M - \num{64770.1} - \num{2760.26} - \num{1315.98} + \num{2.437} R_M = 0 &  &
\end{flalign*}

\begin{equation}
  M_N - M_M + \num{2.437} R_M - \num{68846.34} = 0
  \label{eq:sum_mn_vl2}
\end{equation}

Como o sistema é hiperestático, é necessário utilizar o método das funções de singularidade para obter a equação do momento fletor e integrar duas vezes para obter as equações da declividade e da linha elástica, assim se obtém as equações necessárias para concluir o cálculo das reações de apoio.

Relacionando o momento fletor com a equação da linha elástica temos:

\begin{equation*}
  EI \frac{d^2\nu}{dx^2} = M(x)
\end{equation*}

A função de singularidade do momento fletor é dada por:

\begin{align*}
  EI \frac{d^2\nu}{dx^2} & = R_N \left<x\right>^1 - M_N \left<x\right>^0 -\num{53155.6} \left<x - \num{1.2185}\right>^1 - \num{1580} \left<x - \num{1.747}\right>^1 \rightarrow \\
                         & \quad - \num{540} \left<x - \num{2.437}\right>^1 + R_M \left<x - \num{2.437}\right>^0 + M_M \left<x - \num{2.437}\right>^0
\end{align*}

Resolvendo a primeira integral:

\begin{align*}
  EI \int \frac{d^2\nu}{dx^2} & = \int R_N \left<x\right>^1 - M_N \left<x\right>^0 -\num{53155.6} \left<x - \num{1.2185}\right>^1 - \num{1580} \left<x - \num{1.747}\right>^1 \rightarrow \\
                              & \quad - \num{540} \left<x - \num{2.437}\right>^1 + R_M \left<x - \num{2.437}\right>^1 + M_M \left<x - \num{2.437}\right>^0 \ dx
\end{align*}

\begin{align*}
  EI \frac{d\nu}{dx} & = \frac{R_N}{2} \left<x\right>^2 - M_N \left<x\right>^1 -\frac{\num{53155.6}}{2} \left<x - \num{1.2185}\right>^2 - \frac{\num{1580}}{2} \left<x - \num{1.747}\right>^2 \rightarrow     \\
                     & \quad - \frac{\num{540}}{2} \left<x - \num{2.437}\right>^2 + \frac{R_M}{2} \left<x - \num{2.437}\right>^2 + M_M \left<x - \num{2.437}\right>^1 + C_1 = EI \theta(x) = \left( * \right)
\end{align*}

Resolvendo a segunda integral:

\begin{align*}
  EI \int \frac{d\nu}{dx} & = \int \frac{R_N}{2} \left<x\right>^2 - M_N \left<x\right>^1 -\frac{\num{53155.6}}{2} \left<x - \num{1.2185}\right>^2 - \frac{\num{1580}}{2} \left<x - \num{1.747}\right>^2 \rightarrow \\
                          & \quad - \frac{\num{540}}{2} \left<x - \num{2.437}\right>^2 + \frac{R_M}{2} \left<x - \num{2.437}\right>^2 + M_M \left<x - \num{2.437}\right>^1 + C_1 \ dx
\end{align*}

\begin{align*}
  EI \nu(x) & = \frac{R_N}{6} \left<x\right>^3 - \frac{M_N}{2} \left<x\right>^2 -\frac{\num{53155.6}}{6} \left<x - \num{1.2185}\right>^3 - \frac{\num{1580}}{6} \left<x - \num{1.747}\right>^3 \rightarrow \\
            & \quad - \frac{\num{540}}{6} \left<x - \num{2.437}\right>^3 + \frac{R_M}{6} \left<x - \num{2.437}\right>^3 + \frac{M_M}{2} \left<x - \num{2.437}\right>^2 + C_1 x + C_2 = \left( ** \right)
\end{align*}

Para encontrar as constantes de integração, é necessário aplicar as condições de contorno:

\begin{itemize}
  \item $x = 0$ \hspace{1em} $\theta = 0$ \hspace{1em} Eq. $\left( * \right)$: $\boxed{C_1 = 0}$
  \item $x = 0$ \hspace{1em} $\nu = 0$ \hspace{1em} Eq. $\left( ** \right)$: $\boxed{C_2 = 0}$
  \item $x = \SI{2.437}{\meter}$ \hspace{1em} $\theta = 0$ \hspace{1em} Eq. $\left( * \right)$:
\end{itemize}

\begin{align*}
  0 & = \frac{R_N}{2} \left<\num{2.437}\right>^2 - M_N \left<\num{2.437}\right>^1 -\frac{\num{53155.6}}{2} \left<\num{2.437} - \num{1.2185}\right>^2 - \frac{\num{1580}}{2} \left<\num{2.437} - \num{1.747}\right>^2 \rightarrow \\
    & \quad - \frac{\num{540}}{2} \left<\num{2.437} - \num{2.437}\right>^2 + \frac{R_M}{2} \left<\num{2.437} - \num{2.437}\right>^2 + M_M \left<\num{2.437} - \num{2.437}\right>^1
\end{align*}

\begin{equation*}
  0 = \frac{R_N}{2} \left( \num{2.437} \right)^2 - M_N \left( \num{2.437} \right) -\frac{\num{53155.6}}{2} \left( \num{1.2185} \right)^2 - \frac{\num{1580}}{2} \left( \num{0.69} \right)^2 - \frac{\num{540}}{2} \cdot 0 + \frac{R_M}{2} \cdot 0 + M_M \cdot 0
\end{equation*}

\begin{equation}
  \num{2.97} R_N - \num{2.437} M_N - \num{39837.3} = 0
  \label{eq:cond_contorno_theta_vl2}
\end{equation}

\begin{itemize}
  \item $x = \SI{2.437}{\meter}$ \hspace{1em} $\nu = 0$ \hspace{1em} Eq. $\left( ** \right)$:
\end{itemize}

\begin{align*}
  0 & = \frac{R_N}{6} \left< \num{2.437} \right>^3 - \frac{M_N}{2} \left< \num{2.437} \right>^2 -\frac{\num{53155.6}}{6} \left< \num{2.437} - \num{1.2185} \right>^3 \rightarrow                                      \\
    & \quad - \frac{\num{1580}}{6} \left< \num{2.437} - \num{1.747} \right>^3 - \frac{\num{540}}{6} \left< \num{2.437} - \num{2.437} \right>^3 + \frac{R_M}{6} \left< \num{2.437} - \num{2.437} \right>^3 \rightarrow \\
    & \quad + \frac{M_M}{2} \left< \num{2.437} - \num{2.437} \right>^2
\end{align*}

\begin{align*}

    & \quad - \frac{\num{540}}{6} \cdot 0 + \frac{R_M}{6} \cdot 0 + \frac{M_M}{2} \cdot 0
\end{align*}

\begin{equation}

\end{equation}

Resolvendo as equações \eqref{eq:cond_contorno_theta_vl2} e \eqref{eq:cond_contorno_nu_vl2}:
