A redução necessária é dada pela equação:

\begin{equation}
  I_{\text{nec}} = \frac{n_{\text{motor}}}{n_{\text{tambor}}}
  \label{eq:reducao_necessaria}
\end{equation}

Onde:

\begin{itemize}[leftmargin=1.25cm]
  \item $I_{\text{nec}}$: redução necessária.
  \item $n_{\text{motor}}$: rotação do motor em $\si{\rpm}$ (Tabela \ref{tab:dados_motor_adotado}).
  \item $n_{\text{tambor}}$: rotação do tambor em $\si{\rpm}$.
\end{itemize}

\vspace{1em}

A velocidade do cabo é dada pela equação:

\begin{equation}
  V_{\text{cabo}} = \frac{n_c}{2} \cdot V_L
  \label{eq:velocidade_do_cabo}
\end{equation}

Onde:

\begin{itemize}[leftmargin=1.25cm]
  \item $V_{\text{cabo}}$: velocidade do cabo em $\si{\meter\per\second}$.
  \item $n_c$: número de cabos (Tabela \ref{tab:nc_mmoitao}).
  \item $V_L$: velocidade de elevação em $\si{\meter\per\second}$ (Figura \ref{fig:velocidades}).
\end{itemize}

\vspace{1em}

Substituindo os valores na equação \eqref{eq:velocidade_do_cabo}:

\begin{align*}
  V_{\text{cabo}} & = \frac{4}{2} \cdot 0{,}1               \\
  V_{\text{cabo}} & = \boxed{\SI{0{,}2}{\meter\per\second}}
\end{align*}

A velocidade angular do tambor é dada pela equação:

\begin{equation}
  \omega_{\text{tambor}} = \frac{V_{\text{cabo}}}{R_t}
  \label{eq:velocidade_angular_do_tambor}
\end{equation}

Onde:

\begin{itemize}[leftmargin=1.25cm]
  \item $\omega_{\text{tambor}}$: velocidade angular do tambor em $\si{\radian\per\second}$.
  \item $V_{\text{cabo}}$: velocidade do cabo em $\si{\meter\per\second}$.
  \item $R_t$: raio do tambor em $\si{\meter}$ (Tabela \ref{tab:dados_do_tambor_adotado}).
\end{itemize}

\vspace{1em}

Substituindo os valores na equação \eqref{eq:velocidade_angular_do_tambor}:

\begin{align*}
  \omega_{\text{tambor}} & = \frac{0{,}2}{\frac{355{,}6}{2} \cdot 10^{-3}} \\
  \omega_{\text{tambor}} & = \boxed{\SI{1{,}13}{\radian\per\second}}
\end{align*}

\begin{align*}
  n_{\text{tambor}} & = \frac{60}{2 \pi} \cdot 1{,}13 \\
  n_{\text{tambor}} & = \boxed{\SI{10{,}79}{\rpm}}
\end{align*}

Substituindo os valores na equação \eqref{eq:reducao_necessaria}:

\begin{align*}
  I_{\text{nec}} & = \frac{1170}{10{,}79} \\
  I_{\text{nec}} & = \boxed{108{,}43}
\end{align*}

O torque de acionamento é dado pela equação:

\begin{equation}
  T_1 = \frac{9550 \cdot P_R}{n_{\text{motor}}}
  \label{eq:torque_acionamento}
\end{equation}

Onde:

\begin{itemize}[leftmargin=1.25cm]
  \item $T_1$: torque de acionamento em $\si{\newton\meter}$.
  \item $P_R$: potência requerida em $\si{\kilo\watt}$ \eqref{eq:potencia_requerida}.
  \item $n_{\text{motor}}$: rotação do motor em $\si{\rpm}$ (Tabela \ref{tab:dados_motor_adotado}).
\end{itemize}

\vspace{1em}

Substituindo os valores na equação \eqref{eq:torque_acionamento}:

\begin{align*}
  T_1 & = \frac{9550 \cdot 10{,}96}{1170}     \\
  T_1 & = \boxed{\SI{89{,}46}{\newton\meter}}
\end{align*}

O torque de saída é dado pela equação:

\begin{equation}
  T_2 = T_1 \cdot I_{\text{nec}}
  \label{eq:torque_saida}
\end{equation}

Onde:

\begin{itemize}[leftmargin=1.25cm]
  \item $T_2$: torque de saída em $\si{\newton\meter}$.
  \item $T_1$: torque de acionamento em $\si{\newton\meter}$.
  \item $I_{\text{nec}}$: redução necessária.
\end{itemize}

\vspace{1em}

Substituindo os valores na equação \eqref{eq:torque_saida}:

\begin{align*}
  T_2 & = 89{,}46 \cdot 108{,}43                  \\
  T_2 & = \boxed{\SI{9700{,}15}{\newton\meter}}
\end{align*}

Para o fator de segurança, dado o tempo de funcionamento diário de 4 a 8 horas, adotou-se $FS = 1{,}25$.

\end{align*}

A potência térmica corrigida é dada pela equação:

\begin{equation}
  P_T = P_{T_G} \cdot f_a \cdot f_c \cdot f_h \cdot f_t \cdot f_b
  \label{eq:potencia_termica_corrigida}
\end{equation}
